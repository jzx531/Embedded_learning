    "horse", "sheep", "cow", "elephant", "bear", "zebra", "giraffe", "backpack", "umbrella", "handbag",

    "fork", "knife", "spoon", "bowl", "banana", "apple", "sandwich", "orange", "broccoli", "carrot",

def norm_target(text):
    target = text.strip().lower()
    if target.startswith("voice:"):
        target = target.split(":", 1)[1].strip()
    return target


def letterbox(img, size):
    h, w = img.shape[:2]
    scale = min(size / w, size / h)
    nw, nh = int(w * scale), int(h * scale)
    resized = cv2.resize(img, (nw, nh))
    canvas = np.full((size, size, 3), 114, dtype=np.uint8)
    px, py = (size - nw) // 2, (size - nh) // 2
    canvas[py:py + nh, px:px + nw] = resized
    return canvas, scale, px, py


def detect(net, img, conf_th=0.35, iou_th=0.45):
    inp, scale, px, py = letterbox(img, IMGSZ)
    blob = cv2.dnn.blobFromImage(inp, 1 / 255.0, (IMGSZ, IMGSZ), swapRB=True)
    net.setInput(blob)
    pred = np.squeeze(net.forward()).T
    h, w = img.shape[:2]
    boxes, confs, ids = [], [], []

    for row in pred:
        scores = row[4:]
        cid = int(np.argmax(scores))
        conf = float(scores[cid])
        if conf < conf_th:
            continue
        cx, cy, bw, bh = row[:4]
        x1 = max(0, min(w - 1, int((cx - bw / 2 - px) / scale)))
        y1 = max(0, min(h - 1, int((cy - bh / 2 - py) / scale)))
        x2 = max(0, min(w - 1, int((cx + bw / 2 - px) / scale)))
        y2 = max(0, min(h - 1, int((cy + bh / 2 - py) / scale)))
        if x2 <= x1 or y2 <= y1:
            continue
        boxes.append([x1, y1, x2 - x1, y2 - y1])
        confs.append(conf)
        ids.append(cid)

    keep = cv2.dnn.NMSBoxes(boxes, confs, conf_th, iou_th)
    dets = []
    for i in (np.array(keep).reshape(-1) if len(keep) else []):
        x, y, bw, bh = boxes[i]
        cx, cy = x + bw / 2, y + bh / 2
        dets.append({
            "cx": int(cx),
            "cy": int(cy),
            "dx": int(cx - w / 2),
            "dy": int(cy - h / 2),
            "name": COCO[ids[i]] if ids[i] < len(COCO) else str(ids[i]),
            "conf": confs[i],
            "box": (x, y, x + bw, y + bh),
        })
    dets.sort(key=lambda d: d["conf"], reverse=True)
    return dets


def peak_from_box(det, width):
    if det is None:
        return 0
    left_line = int(round(width * PEAK_LEFT_RATIO))
    right_line = int(round(width * PEAK_RIGHT_RATIO))
    x1, _, x2, _ = det["box"]
    return 1 if x1 <= left_line and x2 >= right_line else 0


def draw_peak_lines(frame):
    h, w = frame.shape[:2]
    left_line = int(round(w * PEAK_LEFT_RATIO))
    right_line = int(round(w * PEAK_RIGHT_RATIO))
    cv2.line(frame, (left_line, 0), (left_line, h), (0, 255, 255), 2)
    cv2.line(frame, (right_line, 0), (right_line, h), (0, 255, 255), 2)


def connect_robot(host, port):
    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    sock.connect((host, port))
    print(f"Vision TCP {host}:{port}")
    return sock


def send_line(sock, line):
    print(line, flush=True)
    if sock:
        try:
            sock.send((line + "\n").encode())
        except OSError as exc:
            print(f"TCP_SEND_FAIL,{exc}", flush=True)


def read_dynamic_target():
    try:
        with open("/tmp/vision_target.txt", "r") as f:
            t = f.read().strip().lower()
            if t and t in COCO:
                return t
    except:
        pass
    return None


def run(target=DEFAULT_TARGET, host="127.0.0.1", port=TCP_PORT, preview=True):
    target = norm_target(target)
    dynamic = (target == "dynamic")
    if not dynamic and target not in COCO:
        print(f"ERROR: unsupported target: {target}")
        return 2

    net = cv2.dnn.readNetFromONNX(MODEL_PATH)
    cap = cv2.VideoCapture(CAMERA_DEV)
    if not cap.isOpened():
        print(f"ERROR: cannot open camera {CAMERA_DEV}")
        return 1
    cap.set(cv2.CAP_PROP_FRAME_WIDTH, CAM_WIDTH)
    cap.set(cv2.CAP_PROP_FRAME_HEIGHT, CAM_HEIGHT)

    sock = None
    auto_started = False
    try:
        sock = connect_robot(host, port)
    except OSError as exc:
        print(f"TCP not connected, stdout only: {exc}")
    last_send = 0.0
    print(f"Vision running target={target}" + (" (dynamic)" if dynamic else ""))
    while True:
        if dynamic:
            new_t = read_dynamic_target()
            if new_t and new_t != target:
                target = new_t
                auto_started = False
                print(f"Vision -> {target}")

        ok, frame = cap.read()
        if not ok:
            time.sleep(0.05)
            continue

        dets = [d for d in detect(net, frame, DETECT_CONF) if d["name"] == target]
        best = dets[0] if dets else None
        peak = peak_from_box(best, frame.shape[1])

        if best:
            line = f"$CAM:{best['cx']},{best['cy']},0,{int(best['conf'] * 100)},{peak}!"
        else:
            line = "$CAM:0,0,0,0,0!"

        now = time.time()
        if now - last_send >= VISION_SEND_INTERVAL:
            send_line(sock, line)
            last_send = now
            if best and not auto_started:
                send_line(sock, "$TASK:START!")
                auto_started = True

        if preview:
            draw_peak_lines(frame)
            if best:
                x1, y1, x2, y2 = best["box"]
                color = (0, 0, 255) if peak else (0, 255, 0)
                cv2.rectangle(frame, (x1, y1), (x2, y2), color, 2)
                label = f"{target} {best['conf']:.2f} PEAK:{peak}"
                cv2.putText(frame, label, (x1, max(22, y1 - 8)), cv2.FONT_HERSHEY_SIMPLEX, 0.55, color, 2)
            cv2.imshow("Vision target PEAK - q to quit", frame)
            if cv2.waitKey(1) & 0xFF == ord("q"):
                break

    cap.release()
    cv2.destroyAllWindows()
    if sock:
        sock.close()
    return 0

def main():
    parser = argparse.ArgumentParser(description="Target-only YOLO vision with PEAK parking output.")
    parser.add_argument("target", nargs="?", default=DEFAULT_TARGET)
    parser.add_argument("--host", default="127.0.0.1")
    parser.add_argument("--port", type=int, default=TCP_PORT)
    parser.add_argument("--no-preview", action="store_true")
    args = parser.parse_args()
    return run(args.target, args.host, args.port, preview=not args.no_preview)


if __name__ == "__main__":
    raise SystemExit(main())
